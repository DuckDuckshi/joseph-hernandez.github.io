	        HAL_GPIO_WritePin(GPIOB, GPIO_PIN_1, GPIO_PIN_RESET);

	        HAL_GPIO_WritePin(GPIOB, GPIO_PIN_14, GPIO_PIN_RESET);
	        HAL_GPIO_WritePin(GPIOB, GPIO_PIN_13, GPIO_PIN_SET);
	        HAL_GPIO_WritePin(GPIOB, GPIO_PIN_1, GPIO_PIN_RESET);
	      }
	      else
	      {
	        HAL_GPIO_WritePin(GPIOB, GPIO_PIN_15, GPIO_PIN_RESET);
	        HAL_GPIO_WritePin(GPIOB, GPIO_PIN_14, GPIO_PIN_RESET);
	        HAL_GPIO_WritePin(GPIOB, GPIO_PIN_13, GPIO_PIN_RESET);
	        HAL_GPIO_WritePin(GPIOB, GPIO_PIN_1, GPIO_PIN_SET);
	      }

	      /* Delay for 500 ms */
	      HAL_Delay(500);
	  Enter_SleepMode();
	  }  //wait again so we don't flood the Serial terminal
    /* USER CODE END WHILE */

    /* USER CODE BEGIN 3 */
  }

void Enter_SleepMode(void)
    {
        // Enable PWR clock
        __HAL_RCC_PWR_CLK_ENABLE();

        // Suspend SysTick interrupt for better power saving
        HAL_SuspendTick();

        // Enter Sleep Mode and wait for interrupt (WFI)
        HAL_PWR_EnterSLEEPMode(PWR_MAINREGULATOR_ON, PWR_SLEEPENTRY_WFI);

        // Resume SysTick interrupt after waking up
        HAL_ResumeTick();
    }
  /* USER CODE END 3 */

float Read_Temperature(void)
{
    uint32_t adc_value = 0;
    float temperature = 0.0;

    // Start ADC Conversion
    HAL_ADC_Start(&hadc3);

    // Poll for ADC conversion completion
    if (HAL_ADC_PollForConversion(&hadc3, HAL_MAX_DELAY) == HAL_OK)
    {
        // Get ADC Value
        adc_value = HAL_ADC_GetValue(&hadc3);

        // Debug: Print ADC value via UART
        char adc_debug[20];
        sprintf(adc_debug, "ADC: %lu\r\n", adc_value);
        HAL_UART_Transmit(&huart2, (uint8_t *)adc_debug, strlen(adc_debug), HAL_MAX_DELAY);

        // Convert ADC Value to Temperature
        temperature = ((float)adc_value * 3300)/(4096 * 10);

        // Debug: Print calculated temperature goes to C room temp should be 68f
        char temp_debug[20];
        sprintf(temp_debug, "Temp: %.2f C\r\n", temperature);
        HAL_UART_Transmit(&huart2, (uint8_t *)temp_debug, strlen(temp_debug), HAL_MAX_DELAY);
    }

    HAL_ADC_Stop(&hadc3);
    return temperature;
}
